<script>
// ==================== SETUP ====================
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const nextCanvas = document.getElementById('next');
const nctx = nextCanvas.getContext('2d');
const COLS = 10, ROWS = 20, SIZE = 30;

let board, piece, nextPiece, score, lines, level, gameOver, paused, running;
let dropInterval, lastDrop, animId;
let audioCtx, masterGain, bgmStarted = false;

// ==================== PIECES ====================
const PIECES = [
  { shape: [[1,1,1,1]], color: '#00f5ff' },           // I - cyan
  { shape: [[1,1],[1,1]], color: '#ffe66d' },          // O - yellow
  { shape: [[0,1,0],[1,1,1]], color: '#a855f7' },      // T - purple
  { shape: [[1,0,0],[1,1,1]], color: '#4ade80' },      // L - green
  { shape: [[0,0,1],[1,1,1]], color: '#f97316' },      // J - orange
  { shape: [[0,1,1],[1,1,0]], color: '#ef4444' },      // S - red
  { shape: [[1,1,0],[0,1,1]], color: '#f43f5e' },      // Z - pink
];

// ==================== PENTATONIC BGM ====================
// Cメジャープentatonic + A minor pentatonic を調和させる
// ルート: A4=440Hz ベースでpentatonicノーツ
const NOTE = {
  // C major pentatonic (C4〜C6) + A minor pentatonic
  'C3': 130.81, 'D3': 146.83, 'E3': 164.81, 'G3': 196.00, 'A3': 220.00,
  'C4': 261.63, 'D4': 293.66, 'E4': 329.63, 'G4': 392.00, 'A4': 440.00,
  'C5': 523.25, 'D5': 587.33, 'E5': 659.25, 'G5': 783.99, 'A5': 880.00,
};

// ペンタトニックインデックス (C D E G A)
const PENTA = [0,2,4,7,9]; // MIDIノートオフセット（メジャースケールから）

function pentatonicNote(degree, octaveShift = 0) {
  const baseOctave = 3 + octaveShift;
  const idx = degree % 5;
  const oct = Math.floor(degree / 5) + baseOctave;
  const notes = ['C','D','E','G','A'];
  const name = notes[idx] + oct;
  return NOTE[name] || 440;
}

// BGMコンテキスト生成
function initAudio() {
  audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  masterGain = audioCtx.createGain();
  masterGain.gain.value = 0.3;
  masterGain.connect(audioCtx.destination);
}

function playTone(freq, duration, type = 'sine', volume = 0.3, delay = 0) {
  if (!audioCtx) return;
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0, audioCtx.currentTime + delay);
  gain.gain.linearRampToValueAtTime(volume, audioCtx.currentTime + delay + 0.02);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + delay + duration);
  osc.connect(gain);
  gain.connect(masterGain);
  osc.start(audioCtx.currentTime + delay);
  osc.stop(audioCtx.currentTime + delay + duration);
}

// メロディパターン（ペンタトニック、不協和音なし）
const MELODY_PATTERNS = [
  // 明るいアルペジオパターン
  [0, 2, 4, 7, 9, 7, 4, 2],
  [0, 4, 9, 7, 4, 2, 0, -1],
  [2, 4, 7, 9, 7, 4, 2, 0],
  [0, 2, 4, 7, 9, 12, 9, 7],
];

let melodyPatternIdx = 0;
let beatCount = 0;
const BPM = 140;
const BEAT_DUR = 60 / BPM;

function scheduleBGM() {
  if (!running || gameOver) return;
  
  const pattern = MELODY_PATTERNS[melodyPatternIdx];
  const baseOctave = Math.floor(Math.random() * 2);
  
  // メロディ（上の声）
  for (let i = 0; i < pattern.length; i++) {
    const deg = pattern[i] + baseOctave * 5;
    if (deg < 0) continue;
    const dur = BEAT_DUR * 0.8;
    playTone(pentatonicNote(deg, 1), dur, 'sine', 0.15, i * BEAT_DUR);
  }

  // バス（下の声）- ルート音のみで完全協和
  const bassNotes = [0, 0, 4, 2]; // C, C, G, E (pentatonic root mapping)
  for (let i = 0; i < 4; i++) {
    playTone(pentatonicNote(bassNotes[i], 0), BEAT_DUR * 1.8, 'triangle', 0.12, i * BEAT_DUR * 2);
  }

  // パッド（和音）- 3度・5度の完全協和音程
  const chordDegrees = [0, 4, 7]; // C, E, G (perfect harmony)
  for (const deg of chordDegrees) {
    playTone(pentatonicNote(deg, 1), BEAT_DUR * 8, 'sine', 0.05, 0);
  }

  melodyPatternIdx = (melodyPatternIdx + 1) % MELODY_PATTERNS.length;
  
  const totalDur = pattern.length * BEAT_DUR;
  setTimeout(scheduleBGM, totalDur * 1000 - 50);
}

function playLineClearSound(count) {
  if (!audioCtx) return;
  const freqs = count >= 4 
    ? [523.25, 659.25, 783.99, 1046.50] 
    : [523.25, 659.25, 783.99];
  freqs.forEach((f, i) => {
    playTone(f, 0.2, 'sine', 0.2, i * 0.08);
  });
}

function playGameOverSound() {
  if (!audioCtx) return;
  [392, 349, 329, 262].forEach((f, i) => {
    playTone(f, 0.4, 'triangle', 0.15, i * 0.2);
  });
}

// ==================== GAME LOGIC ====================
function createBoard() {
  return Array.from({length: ROWS}, () => new Array(COLS).fill(0));
}

function randomPiece() {
  const p = PIECES[Math.floor(Math.random() * PIECES.length)];
  return {
    shape: p.shape.map(r => [...r]),
    color: p.color,
    x: Math.floor(COLS / 2) - Math.floor(p.shape[0].length / 2),
    y: 0,
  };
}

function rotate(shape) {
  const H = shape.length, W = shape[0].length;
  const result = Array.from({length: W}, () => new Array(H).fill(0));
  for (let r = 0; r < H; r++)
    for (let c = 0; c < W; c++)
      result[c][H - 1 - r] = shape[r][c];
  return result;
}

function collides(board, piece) {
  for (let r = 0; r < piece.shape.length; r++) {
    for (let c = 0; c < piece.shape[r].length; c++) {
      if (!piece.shape[r][c]) continue;
      const nx = piece.x + c, ny = piece.y + r;
      if (nx < 0 || nx >= COLS || ny >= ROWS) return true;
      if (ny >= 0 && board[ny][nx]) return true;
    }
  }
  return false;
}

function lock(board, piece) {
  for (let r = 0; r < piece.shape.length; r++) {
    for (let c = 0; c < piece.shape[r].length; c++) {
      if (!piece.shape[r][c]) continue;
      const ny = piece.y + r;
      if (ny < 0) { gameOver = true; return; }
      board[ny][piece.x + c] = piece.color;
    }
  }
}

function clearLines(board) {
  let cleared = 0;
  for (let r = ROWS - 1; r >= 0; r--) {
    if (board[r].every(v => v)) {
      board.splice(r, 1);
      board.unshift(new Array(COLS).fill(0));
      cleared++;
      r++;
    }
  }
  return cleared;
}

function getGhostY(board, piece) {
  let gy = piece.y;
  const tmp = { ...piece, y: gy };
  while (!collides(board, { ...tmp, y: gy + 1 })) gy++;
  return gy;
}

// ==================== RENDERING ====================
function drawBlock(context, x, y, color, size = SIZE, ghost = false) {
  if (ghost) {
    context.strokeStyle = color;
    context.globalAlpha = 0.3;
    context.strokeRect(x * size + 1, y * size + 1, size - 2, size - 2);
    context.globalAlpha = 1;
    return;
  }
  // メイン色
  context.fillStyle = color;
  context.fillRect(x * size + 1, y * size + 1, size - 2, size - 2);
  // ハイライト
  context.fillStyle = 'rgba(255,255,255,0.2)';
  context.fillRect(x * size + 1, y * size + 1, size - 2, 4);
  context.fillRect(x * size + 1, y * size + 1, 4, size - 2);
  // シャドウ
  context.fillStyle = 'rgba(0,0,0,0.2)';
  context.fillRect(x * size + 1, y * size + size - 5, size - 2, 4);
  context.fillRect(x * size + size - 5, y * size + 1, 4, size - 2);
}

function render() {
  // 背景
  ctx.fillStyle = '#16213e';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  // グリッド
  ctx.strokeStyle = 'rgba(255,255,255,0.04)';
  for (let r = 0; r < ROWS; r++)
    for (let c = 0; c < COLS; c++)
      ctx.strokeRect(c * SIZE, r * SIZE, SIZE, SIZE);

  // ボード
  for (let r = 0; r < ROWS; r++)
    for (let c = 0; c < COLS; c++)
      if (board[r][c]) drawBlock(ctx, c, r, board[r][c]);

  if (piece && !gameOver) {
    // ゴースト
    const gy = getGhostY(board, piece);
    for (let r = 0; r < piece.shape.length; r++)
      for (let c = 0; c < piece.shape[r].length; c++)
        if (piece.shape[r][c])
          drawBlock(ctx, piece.x + c, gy + r, piece.color, SIZE, true);

    // 現在のピース
    for (let r = 0; r < piece.shape.length; r++)
      for (let c = 0; c < piece.shape[r].length; c++)
        if (piece.shape[r][c])
          drawBlock(ctx, piece.x + c, piece.y + r, piece.color);
  }

  // Nextプレビュー
  nctx.fillStyle = '#16213e';
  nctx.fillRect(0, 0, nextCanvas.width, nextCanvas.height);
  if (nextPiece) {
    const ns = 20;
    const ox = (nextCanvas.width - nextPiece.shape[0].length * ns) / 2;
    const oy = (nextCanvas.height - nextPiece.shape.length * ns) / 2;
    for (let r = 0; r < nextPiece.shape.length; r++)
      for (let c = 0; c < nextPiece.shape[r].length; c++)
        if (nextPiece.shape[r][c]) {
          nctx.fillStyle = nextPiece.color;
          nctx.fillRect(ox + c * ns, oy + r * ns, ns - 1, ns - 1);
          nctx.fillStyle = 'rgba(255,255,255,0.2)';
          nctx.fillRect(ox + c * ns, oy + r * ns, ns - 1, 3);
        }
  }

  // ゲームオーバー表示
  if (gameOver) {
    ctx.fillStyle = 'rgba(0,0,0,0.7)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = '#e94560';
    ctx.font = 'bold 36px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('GAME OVER', canvas.width / 2, canvas.height / 2 - 10);
    ctx.fillStyle = '#eee';
    ctx.font = '18px sans-serif';
    ctx.fillText(`Score: ${score}`, canvas.width / 2, canvas.height / 2 + 25);
  }

  // ポーズ表示
  if (paused && !gameOver) {
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = '#e94560';
    ctx.font = 'bold 32px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('PAUSE', canvas.width / 2, canvas.height / 2);
  }
}

// ==================== GAME LOOP ====================
function getDropInterval() {
  return Math.max(100, 800 - (level - 1) * 70);
}

function spawnPiece() {
  piece = nextPiece || randomPiece();
  nextPiece = randomPiece();
  if (collides(board, piece)) {
    gameOver = true;
    playGameOverSound();
  }
}

function update() {
  if (gameOver || paused) return;
  
  const now = Date.now();
  if (now - lastDrop > dropInterval) {
    const moved = { ...piece, y: piece.y + 1 };
    if (!collides(board, moved)) {
      piece.y++;
    } else {
      lock(board, piece);
      if (!gameOver) {
        const cleared = clearLines(board);
        if (cleared > 0) {
          const pts = [0, 100, 300, 500, 800];
          score += pts[cleared] * level;
          lines += cleared;
          level = Math.floor(lines / 10) + 1;
          document.getElementById('score').textContent = score;
          document.getElementById('lines').textContent = lines;
          document.getElementById('level').textContent = level;
          playLineClearSound(cleared);
        }
        spawnPiece();
      }
    }
    lastDrop = now;
  }
}

function gameLoop() {
  update();
  render();
  animId = requestAnimationFrame(gameLoop);
}

// ==================== INPUT ====================
document.addEventListener('keydown', e => {
  if (gameOver) return;
  
  if (e.key === 'p' || e.key === 'P') {
    paused = !paused;
    return;
  }
  if (paused) return;

  switch (e.key) {
    case 'ArrowLeft': {
      const m = { ...piece, x: piece.x - 1 };
      if (!collides(board, m)) piece.x--;
      break;
    }
    case 'ArrowRight': {
      const m = { ...piece, x: piece.x + 1 };
      if (!collides(board, m)) piece.x++;
      break;
    }
    case 'ArrowDown': {
      const m = { ...piece, y: piece.y + 1 };
      if (!collides(board, m)) { piece.y++; score += 1; }
      lastDrop = Date.now();
      break;
    }
    case 'ArrowUp': {
      const rotated = { ...piece, shape: rotate(piece.shape) };
      // ウォールキック（簡易）
      const kicks = [0, -1, 1, -2, 2];
      for (const k of kicks) {
        const wk = { ...rotated, x: rotated.x + k };
        if (!collides(board, wk)) { piece.shape = wk.shape; piece.x = wk.x; break; }
      }
      break;
    }
    case ' ': {
      // ハードドロップ
      while (!collides(board, { ...piece, y: piece.y + 1 })) {
        piece.y++; score += 2;
      }
      const m = { ...piece, y: piece.y + 1 };
      if (!collides(board, m)) { /* already at bottom */ }
      lock(board, piece);
      if (!gameOver) {
        const cleared = clearLines(board);
        if (cleared > 0) {
          const pts = [0, 100, 300, 500, 800];
          score += pts[cleared] * level;
          lines += cleared;
          level = Math.floor(lines / 10) + 1;
          document.getElementById('score').textContent = score;
          document.getElementById('lines').textContent = lines;
          document.getElementById('level').textContent = level;
          playLineClearSound(cleared);
        }
        spawnPiece();
      }
      lastDrop = Date.now();
      break;
    }
  }
});

// ==================== START / TOGGLE ====================
function toggleGame() {
  const btn = document.getElementById('startBtn');
  if (!running) {
    if (!audioCtx) initAudio();
    if (audioCtx.state === 'suspended') audioCtx.resume();
    
    board = createBoard();
    score = 0; lines = 0; level = 1;
    gameOver = false; paused = false;
    document.getElementById('score').textContent = 0;
    document.getElementById('lines').textContent = 0;
    document.getElementById('level').textContent = 1;
    
    nextPiece = randomPiece();
    spawnPiece();
    
    dropInterval = getDropInterval();
    lastDrop = Date.now();
    
    running = true;
    btn.textContent = 'RESTART';
    
    if (!bgmStarted) { bgmStarted = true; scheduleBGM(); }
    
    if (animId) cancelAnimationFrame(animId);
    gameLoop();
  } else {
    // リスタート
    if (animId) cancelAnimationFrame(animId);
    running = false;
    toggleGame();
  }
}
</script>
</body>
</html>

## 機能まとめ

| 機能 | 説明 |
|------|------|
| **操作** | `←` `→` 移動 / `↑` 回転 / `↓` 高速落下 / `Space` ハードドロップ / `P` ポーズ |
| **BGM** | Cメジャープentatonic（C-D-E-G-A）のみを使用 → 常に協和した心地よい音 |
| **SE** | ライン消去時に上昇するアルペジオ / ゲームオーバー時に下降するメロディ |
| **機能** | ゴースト表示、Next表示、ウォールキック、レベルアップで速度UP |

BGMは **ペンタトニックスケール（5音音階）** を使っているため、どの音を重ねても不協和音が生まれません。メロディ・バス・パッドの3層構造で豊かな響きを実現しています。
